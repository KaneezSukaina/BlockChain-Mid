def main(argv):
    if len(argv) < 2:
        usage(); return 2
    cmd = argv[1].lower()
    try:
        if cmd == "setup": setup()
        elif cmd == "demo": demo()
        elif cmd == "status": status()
        elif cmd == "simulate" and len(argv) >= 3: simulate(argv[2])
        elif cmd == "analyse": analyse(argv[2] if len(argv) >= 3 else "RSU1")
        elif cmd == "enforce": enforce()
        elif cmd == "access" and len(argv) >= 3:
            vehicle_id = argv[2].upper()
            operation = (
                argv[3] if len(argv) >= 4
                else "protected_operation"
            )
            allowed = access_control(vehicle_id, operation)
            if not allowed:
                print("Protected operation BLOCKED.")
        elif cmd == "bench" and len(argv) >= 3: bench(argv[2])
        else: usage(); return 2
    except KeyboardInterrupt:
        print("\nStopped by user."); return 130
    except Exception as exc:
        print(f"\nERROR: {exc}", file=sys.stderr)
        print("Check that all four MultiChain daemons are running and their RPC ports/config files match this script.", file=sys.stderr)
        return 1
    return 0
